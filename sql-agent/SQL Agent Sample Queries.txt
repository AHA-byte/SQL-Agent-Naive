**1** shows current QBCC reminders and the scope amounts for QBCC + what QBCC has been purchased and the QBCC policy number


SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

CREATE view [admin].[qbcc] as 
  with cte_job as (
    select 
      j.id as jobid
      ,j.jobnumber
      ,min(r.title) as Reminder
      ,cast(min(r.createdAt)as date) as ReminderDate
    FROM t_jobs j
    join t_statuses s on s.id = j.statusId
    left join [dbo].[t_reminders] r on j.id = r.jobid and r.title like '%qbcc%' and r.reminderStatus not in ('completed','cancelled')    
    where s.statusType not in ('Booking','Quoting','Submission','Pending Approval','Contract','Closed')
        and not (s.statusType = 'Accounts' and s.[description] = 'Invoiced')
    group by j.id,jobnumber
    
    ),
    cte_sow as (
    select    
        j.id as jobid
        ,j.jobnumber
        ,round(sum(es.totalIncludingTax),0) as SOW
        from [t_estimates-snapshot] es 
        join t_jobs j on es.jobid = j.id 
        join t_statuses s on s.id = j.statusId 
        where es.estimateType = 'Authorised Works' 
            and es.estimateStatus like '%auth%'
            and s.statusType not in ('Booking','Quoting','Submission','Pending Approval','Contract','Closed')
            and not (s.statusType = 'Accounts' and s.[description] = 'Invoiced')
        group by j.id,j.jobnumber
        ),
    cte_itm as (
    SELECT 
        es.jobId,
        SUM(
            CASE 
            WHEN ISNULL(esi.labourTotal, 0) > 0 THEN ISNULL(esi.labourTotal, 0)
            WHEN ISNULL(esi.materialTotal, 0) > 0 THEN ISNULL(esi.materialTotal, 0)
            ELSE 0
            END
        ) AS [SOW QBCC]
        FROM [dbo].[t_estimates-snapshot-items] esi
        join [dbo].[t_estimates-snapshot-categories] esc on esc.categoryid = esi.categoryid
        JOIN [t_estimates-snapshot] es ON es.id = esi.estimateId
        WHERE 
        lower(esi.description) LIKE '%qbcc%'
        and (esc.label in ('general','Insurances','preliminaries'))
        AND es.estimateType = 'Authorised Works'
        GROUP BY es.jobId
    ),
    cte_exp as (
    select    
        j.id as jobid
        ,round(sum(exp.amount),2) as 'QBCC Purch'
        from t_jobs j
        join t_statuses s on s.id = j.statusId 
        join [t_job-expense-invoices] exp on exp.jobid = j.id and exp.description like '%qbcc%'
        where s.statusType not in ('Booking','Quoting','Submission','Pending Approval','Contract','Closed')
        and not (s.statusType = 'Accounts' and s.[description] = 'Invoiced')
        group by j.id
        ),
    cte_qbcc as (
    SELECT
        q.jobid
        ,q.[policy number]
        ,q.[contract amount]
        ,q.[premium amount]
        from [admin].[qbcc-policies] q 
        join t_jobs j on q.jobid = j.id
        join t_statuses s on s.id = j.statusId 
        where q.status <> 'cancelled' 
        and s.statusType not in ('Booking','Quoting','Submission','Pending Approval','Contract','Closed')
        and not (s.statusType = 'Accounts' and s.[description] = 'Invoiced')
    )
  select r.jobnumber
        ,r.Reminder
        ,r.ReminderDate
        ,s.SOW
        ,i.[SOW QBCC]
        ,(s.SOW - 
            CASE WHEN i.[SOW QBCC] IS NULL THEN 0 ELSE i.[SOW QBCC]
            END) as 'SOW ex QBCC'
        ,e.[QBCC Purch]
        ,q.[policy number]
        ,q.[contract amount]
        ,q.[premium amount]
  from cte_job r
  left join cte_exp e on r.jobid = e.jobid
  left join cte_sow s on r.jobid = s.jobid
  left join cte_itm i on r.jobid = i.jobid
  left join cte_qbcc q on r.jobid = q.jobid
  
  WHERE (s.SOW - 
    CASE WHEN i.[SOW QBCC] IS NULL THEN 0 ELSE i.[SOW QBCC]
    END) > 3300;
GO


**2** reminders for current month and average days to complete that reminder and average # reminders assigned for each day

SELECT  
    u.fullName,

    -- Only completed reminders contribute to this average
    CAST(
        AVG(
            CASE 
                WHEN r.reminderstatus = 'Completed'
                    THEN DATEDIFF(DAY, r.createdAt, r.updatedAt)
                ELSE NULL
            END
        ) AS DECIMAL(10,1)
    ) AS [Average Days to Complete],

    -- All reminders count toward claims/day
    CAST(
        CAST(COUNT(*) AS FLOAT)
        / NULLIF(COUNT(DISTINCT CAST(r.createdAt AS DATE)), 0)
        AS DECIMAL(10,1)
    ) AS [Average Claims Per Day]

FROM t_reminders r
JOIN t_jobs j 
    ON r.jobid = j.id
JOIN t_contacts c 
    ON c.id = j.clientId
LEFT JOIN clients cl 
    ON cl.[Instructing Client] = c.name
JOIN t_users u 
    ON r.userId = u.id

WHERE r.createdAt >= '2026-02-01'

GROUP BY u.fullName
ORDER BY u.fullName;

CREATE VIEW [dbo].[Allocations] AS

WITH AP_Totals AS (
    SELECT
        ap.workOrderId,
        SUM(ap.subTotal) AS TotalSubTotal
    FROM [t_accounts-payable-invoices] ap
    WHERE ap.approvalStatus NOT IN ('cancelled')
    GROUP BY ap.workOrderId
)

    j.[jobNumber] as "Job Number",
    j.tags as "Tags",
    a.[allocationNumber] as "Allocation Number",
    a.[allocationType] as "Type",
    st.[label] as "Status",
    a.label as "Label",
    a.[createdBy] as "Created By",
    a.[createdAt] as "Allocation Created",
    ac.name as "Assigned Contact",
    ai.fullName as "Assigned User",
    a.[firstCustomerContactAt] as "First Customer Contact At",
    a.siteAttended as "Start / Site Attended",
    a.[completed] as "Completed",
    wo.label as "Work Order Number",
    woa.name as "Work Order Assigned",
    ap.TotalSubTotal as "Accounts Payable Invoices Total (Exl. Tax)",
    a.updatedAt as "UpdatedAt"
FROM [dbo].[t_allocations] a
LEFT JOIN [t_work-orders] wo ON wo.allocationId = a.id AND wo.workOrderStatus NOT IN ('Cancelled', 'Draft')
LEFT JOIN AP_Totals ap ON ap.workOrderId = wo.id
LEFT JOIN t_jobs j ON j.id = a.jobId
LEFT JOIN [t_allocation-statuses] st ON st.id = a.allocationStatusId
LEFT JOIN [t_contacts] ac ON ac.id = a.assignedContactId
LEFT JOIN [t_users] ai ON ai.id = a.assignedId
LEFT JOIN [t_contacts] woa ON woa.id = wo.assignedId

**6** used to calculate the historic % split of authorised works by range bands over past few months

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE VIEW [model].[bandsplit] AS

-- Step 1: Group estimates per job to get total SOW per job
WITH job_sow AS (
    SELECT 
        j.id AS JobId,
        j.jobNumber,
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],
        ROUND(SUM(es.totalIncludingTax) / 1.1, 2) AS SOW
    FROM [dbo].[t_estimates-snapshot] es
    JOIN t_jobs j ON j.id = es.jobId
    JOIN [t_catastrophe-codes] cc ON cc.id = j.catastropheCodeId
    JOIN t_contacts c ON c.id = j.clientId
    JOIN clients cl ON cl.[Instructing Client] = c.name
    JOIN t_statuses s ON s.id = j.statusId
    WHERE 
        es.estimateType = 'Authorised Works'
        AND es.estimateStatus IN ('Authorised','Authorised Variation')
        AND NOT (es.label LIKE '%lq%' OR es.label LIKE '%liability%') 
        AND j.jobNumber NOT LIKE '%test%'
        AND j.createdAt > EOMONTH(GETDATE(), -24)
        AND cl.[Client Group] NOT IN ('Commercial','Internal')
    GROUP BY 
        j.id, j.jobNumber, cc.code, cl.[Client Group]
),

-- Step 2: Categorize each job by band
scoped_data AS (
    SELECT 
        [BAU/CAT],
        CASE 
            WHEN SOW <= 0 THEN ''
            WHEN SOW < 5000 THEN '<5000'
            WHEN SOW < 15000 THEN '<15000'
            WHEN SOW < 30000 THEN '<30000'
            WHEN SOW < 50000 THEN '<50000'
            WHEN SOW < 100000 THEN '<100000'
            ELSE '>100000'
        END AS Band,
        SOW
    FROM job_sow
),

-- Step 3: Aggregate per band
band_stats AS (
    SELECT 
        [BAU/CAT],
        Band,
        COUNT(*) AS JobCount,
        SUM(SOW) AS TotSOW,
        ROUND(AVG(SOW), -2) AS AvgSOW
    FROM scoped_data
    GROUP BY [BAU/CAT], Band
),

-- Step 4: Total jobs per BAU/CAT
bau_totals AS (
    SELECT 
        [BAU/CAT],
        COUNT(*) AS TotalJobs
    FROM scoped_data
    GROUP BY [BAU/CAT]
)

-- Step 5: Final output
SELECT 
    b.[BAU/CAT],
    b.Band,
    b.JobCount,
    b.TotSOW,
    b.AvgSOW,
    CAST(b.JobCount * 100.0 / bt.TotalJobs AS DECIMAL(5,2)) AS [Band % Share]
FROM band_stats b
JOIN bau_totals bt ON b.[BAU/CAT] = bt.[BAU/CAT]
where band <> '';
GO

**9** works out the distribution split of jobs over bands over recent historic months

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

CREATE VIEW [model].[distribution] AS
WITH cte_sow AS (
    SELECT 
        jobId,
        SUM(ROUND(totalIncludingTax / 1.1, 2)) AS SOW
    FROM [dbo].[t_estimates-snapshot]
    WHERE 
        estimateStatus IN ('Authorised', 'Authorised Variation')
        AND estimateType = 'Authorised Works'
        AND NOT (label LIKE '%lq%' OR label LIKE '%liability%')
    GROUP BY jobId
),
cte_ar AS (
    SELECT 
        jobId,
        invoicedDate,
        subtotal
    FROM [t_accounts-receivable-invoices]
    WHERE 
        accountNumber BETWEEN '201' AND '240'
        AND accountNumber NOT IN ('212', '213', '214')
        AND invoicedDate IS NOT NULL
),
cte_status_start AS (
    SELECT 
        jobId,
     --   MIN(CASE WHEN s.statusType = 'Booking' THEN sh.createdAt END) AS BookingStart,
        MIN(CASE WHEN s.statusType = 'Pending Approval' THEN sh.createdAt END) AS PendingStart,
        MIN(CASE WHEN s.statusType = 'Contract' THEN sh.createdAt END) AS ContractStart,
        MIN(CASE WHEN s.statusType = 'Construction' THEN sh.createdAt END) AS ConstructionStart
    FROM [t_status-histories] sh
    JOIN t_statuses s ON s.id = sh.newStatusId
    WHERE 
        sh.createdAt > EOMONTH('2025-04-30', -18)
    GROUP BY jobId
),
cte_base AS (
    SELECT 
        j.id AS jobId,
        j.createdAt AS jobCreated,
        ar.invoicedDate,
        ar.subtotal,
        sow.SOW,
    --    ss.BookingStart,
        ss.PendingStart,
        ss.ContractStart,
        ss.ConstructionStart,

        CASE 
            WHEN sow.SOW > 0 THEN ar.subtotal / sow.SOW
            ELSE 0
        END AS Weight,

        CASE 
            WHEN sow.SOW <= 0 THEN ''
            WHEN sow.SOW < 5000 THEN '<5000'
            WHEN sow.SOW < 15000 THEN '<15000'
            WHEN sow.SOW < 30000 THEN '<30000'
            WHEN sow.SOW < 50000 THEN '<50000'
            WHEN sow.SOW < 100000 THEN '<100000'
            ELSE '>100000'
        END AS Band,

        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],

        round(DATEDIFF(DAY, j.createdAt, ar.invoicedDate) / 30.0,0) AS BookingMths,
        round(DATEDIFF(DAY, ss.PendingStart, ar.invoicedDate) / 30.0,0) AS PendingMths,
        round(DATEDIFF(DAY, ss.ContractStart, ar.invoicedDate) / 30.0,0) AS ContractMths,
        round(DATEDIFF(DAY, ss.ConstructionStart, ar.invoicedDate) / 30.0,0) AS ConstructionMths

    FROM cte_ar ar
    JOIN [dbo].[t_jobs] j ON ar.jobId = j.id
    JOIN [t_catastrophe-codes] cc ON cc.id = j.catastropheCodeId
    JOIN t_contacts c ON c.id = j.clientId
    JOIN clients cl ON cl.[Instructing Client] = c.name
    JOIN cte_sow sow ON sow.jobId = j.id
    JOIN cte_status_start ss ON ss.jobId = j.id
    WHERE 
        j.createdAt BETWEEN EOMONTH(GETDATE(), -18) AND EOMONTH(GETDATE(), -6)
),
cte_cleaned AS (
    SELECT
        [BAU/CAT],
        Band,
        CASE --WHEN BookingMths < 1 THEN 1 
             WHEN BookingMths > 15 THEN 15 
             ELSE BookingMths 
        END AS BookingMth,
        CASE --WHEN PendingMths < 1 THEN 1 
             WHEN PendingMths > 15 THEN 15 
             ELSE PendingMths 
        END AS PendingMth,
        CASE --WHEN ContractMths < 1 THEN 1 
             WHEN ContractMths > 15 THEN 15 
             ELSE ContractMths 
        END AS ContractMth,
        CASE --WHEN ConstructionMths < 1 THEN 1 
             WHEN ConstructionMths > 15 THEN 15 
             ELSE ConstructionMths 
        END AS ConstructionMth,
        Weight
    FROM cte_base
    WHERE Band <> ''
),
numbers AS (
       SELECT 0 AS Mths UNION ALL SELECT 1 UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL SELECT 4 UNION ALL SELECT 5
 --   SELECT 1 AS Mths UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL SELECT 4 UNION ALL SELECT 5
    UNION ALL SELECT 6 UNION ALL SELECT 7 UNION ALL SELECT 8 UNION ALL SELECT 9 UNION ALL SELECT 10
    UNION ALL SELECT 11 UNION ALL SELECT 12 UNION ALL SELECT 13 UNION ALL SELECT 14 UNION ALL SELECT 15
)

SELECT 
    c.[BAU/CAT],
    c.Band,
    n.Mths,
    CAST(SUM(CASE WHEN c.BookingMth = n.Mths THEN Weight ELSE 0 END) / NULLIF(SUM(Weight), 0) AS DECIMAL(6,5)) AS [Booking %],
    CAST(SUM(CASE WHEN c.PendingMth = n.Mths THEN Weight ELSE 0 END) / NULLIF(SUM(Weight), 0) AS DECIMAL(6,5)) AS [Pending %],
    CAST(SUM(CASE WHEN c.ContractMth = n.Mths THEN Weight ELSE 0 END) / NULLIF(SUM(Weight), 0) AS DECIMAL(6,5)) AS [Contract %],
    CAST(SUM(CASE WHEN c.ConstructionMth = n.Mths THEN Weight ELSE 0 END) / NULLIF(SUM(Weight), 0) AS DECIMAL(6,5)) AS [Construction %]
FROM cte_cleaned c
CROSS JOIN numbers n
GROUP BY c.[BAU/CAT], c.Band, n.Mths;
GO

            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],
        CASE 
            WHEN REPLACE(LOWER(LEFT(REPLACE(cc.label, ' ', ''), 6)), ' ', '') LIKE '%nocata%' THEN 'BAU'
            WHEN REPLACE(LOWER(LEFT(REPLACE(cc.label, ' ', ''), 6)), ' ', '') LIKE 'bau%' THEN 'BAU'
            WHEN REPLACE(LOWER(LEFT(REPLACE(cc.label, ' ', ''), 6)), ' ', '') LIKE '%bus%' THEN 'BAU'
            WHEN REPLACE(LOWER(LEFT(REPLACE(cc.label, ' ', ''), 6)), ' ', '') LIKE '%cyc%' THEN 'CATDEB'
            ELSE LEFT(REPLACE(LEFT(REPLACE(cc.label, ' ', ''), 6), ' ', ''), 6)
        END AS CatCode,
        s.statusType,
        s.name AS status
    FROM t_jobs j
    JOIN [t_catastrophe-codes] cc ON j.catastropheCodeId = cc.id
    JOIN t_contacts c ON c.id = j.clientId
    LEFT JOIN clients cl ON cl.[instructing client] = c.name
    JOIN t_statuses s ON s.id = j.statusId
    WHERE 
        j.jobNumber NOT LIKE '%test%'
        AND cl.[Client Group] NOT IN ('Commercial','Internal')
        AND s.statusType = 'Pending Approval'
),
cte_sow AS (
    SELECT 
        CASE 
            WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' 
            ELSE cl.[Client Group] 
        END AS [Client Group],
        es.[jobId],
        j.jobNumber,
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],
    --    es.[label],
        s.statusType,
        s.name AS status,
        --ROUND(es.totalIncludingTax / 1.1, 2) AS SOW
        SUM(ROUND(es.totalIncludingTax / 1.1, 2)) AS SOW
    FROM [dbo].[t_estimates-snapshot] es
    JOIN t_jobs j ON j.id = es.jobId
    JOIN [t_catastrophe-codes] cc ON j.catastropheCodeId = cc.id
    JOIN t_contacts c ON c.id = j.clientId
    LEFT JOIN clients cl ON cl.[instructing client] = c.name
    JOIN t_statuses s ON s.id = j.statusId
    WHERE 
        (es.estimateStatus = 'pending' AND NOT (es.label LIKE '%lq%' OR es.label LIKE '%liability%') OR es.estimateStatus = 'authorised')
        AND j.jobNumber NOT LIKE '%test%'
        AND cl.[Client Group] <> 'Commercial'
        AND s.statusType = 'Pending Approval'
        AND es.estimateType = 'Authorised Works'
      --  AND es.jobId NOT IN (                                Leave out as should be summing all pending + auth SOW for that job
      --     SELECT DISTINCT jobId
      --     FROM [dbo].[t_estimates-snapshot]
      --     WHERE 
      --         estimateStatus = 'Authorised'
      --         AND estimateType = 'Authorised Works'
      --         AND NOT (label LIKE '%lq%' OR label LIKE '%liability%'))
    GROUP BY 
        cl.[Client Group], es.jobId, j.jobNumber, cc.code, s.statusType, s.name
),
cte_start AS (
    SELECT 
        jobId,
        MIN(CASE WHEN s.statusType = 'Pending Approval' THEN cast(sh.createdAt as date) END) AS PendingStart
    FROM [t_status-histories] sh
    JOIN t_statuses s ON s.id = sh.newStatusId
    GROUP BY jobId
)

SELECT 
    j.[Client Group],
    j.jobId,
    j.jobNumber,
    j.[BAU/CAT],
    j.CatCode,
    cte_start.PendingStart,
   -- cte_sow.label,
    j.statusType,
    j.status,
    cte_sow.SOW,
    CASE 
        WHEN cte_sow.SOW IS NULL THEN 'None'
        WHEN cte_sow.SOW <= 0 THEN ''
        WHEN cte_sow.SOW < 5000 THEN '<5000'
        WHEN cte_sow.SOW < 15000 THEN '<15000'
        WHEN cte_sow.SOW < 30000 THEN '<30000'
        WHEN cte_sow.SOW < 50000 THEN '<50000'
        WHEN cte_sow.SOW < 100000 THEN '<100000'
        WHEN cte_sow.SOW > 100000 THEN '>100000'
        ELSE 'none'
    END AS Band
FROM cte_jobs j
LEFT JOIN cte_start ON j.jobid = cte_start.jobid
LEFT JOIN cte_sow ON j.jobid = cte_sow.jobid;

**11** calculates win rates for jobs from historic averages

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

CREATE VIEW [model].[winrates] AS
WITH cte_jobs AS (
    SELECT 
        j.id AS jobId,
        j.jobNumber,
        j.createdAt,
        --p.label AS Peril,
        CASE 
            WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' 
            ELSE cl.[Client Group] 
        END AS [Client Group],
        cc.code AS catCode,
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT]
    FROM [dbo].[t_jobs] j
    --JOIN t_perils p ON p.id = j.perilId
    JOIN [t_catastrophe-codes] cc ON cc.id = j.catastropheCodeId
    JOIN t_statuses s ON s.id = j.statusId
    JOIN t_contacts c ON c.id = j.clientId
    JOIN clients cl ON cl.[Instructing Client] = c.name
    WHERE 
        j.createdAt BETWEEN EOMONTH(GETDATE(), -18) AND EOMONTH(GETDATE(), -6)
        AND cc.code <> 'Commercial'
        AND cl.[Client Group] NOT IN ('Commercial', 'Internal')
),
cte_status_flags AS (
    SELECT
        j.id AS jobId,
        MAX(CASE WHEN s.statusType = 'Booking' THEN 1 ELSE 0 END) AS Booking,
        MAX(CASE WHEN s.statusType = 'Pending Approval' THEN 1 ELSE 0 END) AS Pending,
        MAX(CASE WHEN s.statusType = 'Contract' THEN 1 ELSE 0 END) AS Contracts,
        MAX(CASE WHEN s.statusType = 'Construction' THEN 1 ELSE 0 END) AS Apprvd
    FROM [t_status-histories] sh
    JOIN t_jobs j ON j.id = sh.jobId
    JOIN t_statuses s ON s.id = sh.newStatusId
    WHERE j.createdAt BETWEEN EOMONTH(GETDATE(), -18) AND EOMONTH(GETDATE(), -6)
    GROUP BY j.id
),
cte_full AS (
    SELECT 
        --j.peril,
        j.[Client Group],
        j.[BAU/CAT],
        ISNULL(f.Booking, 0) AS Booking,
        ISNULL(f.Pending, 0) AS Pending,
        ISNULL(f.Contracts, 0) AS Contracts,
        ISNULL(f.Apprvd, 0) AS Apprvd,
        -- Determine win outcome per job
        CASE 
            WHEN ISNULL(f.Apprvd, 0) = 1 AND ISNULL(f.Booking, 0) = 1 THEN 1 
            ELSE 0 
        END AS BookingWin,

        CASE 
            WHEN ISNULL(f.Apprvd, 0) = 1 AND ISNULL(f.Pending, 0) = 1 THEN 1 
            ELSE 0 
        END AS PendingWin,

        CASE 
            WHEN ISNULL(f.Apprvd, 0) = 1 AND ISNULL(f.Contracts, 0) = 1 THEN 1 
            ELSE 0 
        END AS ContractsWin
    FROM cte_jobs j
    LEFT JOIN cte_status_flags f ON j.jobId = f.jobId
)

SELECT 
    [BAU/CAT],
    [Client Group],
    --Peril,

    COUNT(CASE WHEN Booking = 1 THEN 1 END) AS [Booking Count],
    COUNT(CASE WHEN Pending = 1 THEN 1 END) AS [Pending Count],
    COUNT(CASE WHEN Contracts = 1 THEN 1 END) AS [Contracts Count],

    CAST(SUM(BookingWin) * 1.0 / NULLIF(COUNT(CASE WHEN Booking = 1 THEN 1 END), 0) AS DECIMAL(5,2)) AS [Booking Win %],
    CAST(SUM(PendingWin) * 1.0 / NULLIF(COUNT(CASE WHEN Pending = 1 THEN 1 END), 0) AS DECIMAL(5,2)) AS [Pending Win %],
    CAST(SUM(ContractsWin) * 1.0 / NULLIF(COUNT(CASE WHEN Contracts = 1 THEN 1 END), 0) AS DECIMAL(5,2)) AS [Contracts Win %]

FROM cte_full
GROUP BY 
    [BAU/CAT],
    [Client Group];
    --peril;
GO


**12** historic win rates for jobs in preconstruction (status type booking, quoting, submission, pending approval, contract)

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

CREATE VIEW [model].[precon_winrates] AS
WITH cte_jobs AS (
    SELECT 
        CASE 
            WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' 
            ELSE cl.[Client Group] 
        END AS [Client Group],
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],
        COUNT(j.id) AS TotalClaims
    FROM [dbo].[t_jobs] j
    JOIN [t_catastrophe-codes] cc ON cc.id = j.catastropheCodeId
    JOIN t_statuses s ON s.id = j.statusId
    JOIN t_contacts c ON c.id = j.clientId
    JOIN clients cl ON cl.[Instructing Client] = c.name
    WHERE 
        j.createdAt BETWEEN EOMONTH(GETDATE(), -18) AND EOMONTH(GETDATE(), -6)
        AND cc.code <> 'Commercial'
        AND cl.[Client Group] NOT IN ('Commercial', 'Internal')
    GROUP BY 
        CASE 
            WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' 
            ELSE cl.[Client Group] 
        END,
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END
),
cte_alloc_flags AS (
    SELECT
        j.id AS jobId,
        CASE 
            WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' 
            ELSE cl.[Client Group] 
        END AS [Client Group],
        CASE 
            WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
            WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
            ELSE 'CAT'
        END AS [BAU/CAT],
        MAX(CASE WHEN a.allocationType = 'Assessment Report' THEN 1 ELSE 0 END) AS AF,
        MAX(CASE WHEN a.allocationType = 'Make Safe' THEN 1 ELSE 0 END) AS MS,
        MAX(CASE WHEN a.allocationType NOT IN ('Assessment Report', 'Make Safe', 'Restoration') THEN 1 ELSE 0 END) AS SF
    FROM [dbo].[t_Allocations] a
    JOIN [t_allocation-statuses] ast ON a.allocationStatusId = ast.id
    JOIN [dbo].[t_jobs] j ON a.jobId = j.id
    JOIN [t_catastrophe-codes] cc ON cc.id = j.catastropheCodeId
    JOIN t_statuses s ON s.id = j.statusId
    JOIN t_contacts c ON c.id = j.clientId
    JOIN clients cl ON cl.[Instructing Client] = c.name
    WHERE
        ast.label NOT IN (
            'Cancelled',
            'Draft',
            'Allocation Returned',
            'Cancelled - WMC Rectification',
            'Not Fulfilled - No invoice Required',
            'No Invoice Required',
            'Quote - no invoice required',
            'Re-attendance - No Invoice Required'
        )
        AND j.createdAt BETWEEN EOMONTH(GETDATE(), -18) AND EOMONTH(GETDATE(), -6)
        AND cc.code <> 'Commercial'
        AND cl.[Client Group] NOT IN ('Commercial', 'Internal')
    GROUP BY j.id,
             CASE WHEN cl.[Client Group] = 'Suncorp' THEN 'AAI Ltd' ELSE cl.[Client Group] END,
             CASE WHEN cc.code IN ('BAU','NO-CAT') THEN 'BAU'
                  WHEN cl.[Client Group] = 'Commercial' THEN 'COM'
                  ELSE 'CAT' END
),
cte_count AS (
    SELECT 
        [Client Group],
        [BAU/CAT],
        COUNT(CASE WHEN AF = 1 THEN 1 END) AS [AF Count],
        COUNT(CASE WHEN MS = 1 THEN 1 END) AS [MS Count],
        COUNT(CASE WHEN SF = 1 THEN 1 END) AS [SF Count]
    FROM cte_alloc_flags
    GROUP BY [Client Group], [BAU/CAT]
)

SELECT 
    j.[BAU/CAT],
    j.[Client Group],
    j.TotalClaims,
    c.[AF Count],
    c.[MS Count],
    c.[SF Count],
    CAST(c.[AF Count] * 1.0 / NULLIF(j.TotalClaims, 0) AS DECIMAL(5,2)) AS [AF Win %],
    CAST(c.[MS Count] * 1.0 / NULLIF(j.TotalClaims, 0) AS DECIMAL(5,2)) AS [MS Win %],
    CAST(c.[SF Count] * 1.0 / NULLIF(j.TotalClaims, 0) AS DECIMAL(5,2)) AS [SF Win %]
FROM cte_jobs j
left JOIN cte_count c 
    ON j.[Client Group] = c.[Client Group] AND j.[BAU/CAT] = c.[BAU/CAT];
GO


**13** stored proc to return various key data items for a finals workbook from the estimates and work orders

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE PROCEDURE [reporting].[FinalsWOData] (@JobNumber NVARCHAR(50))

AS

BEGIN
    SET NOCOUNT ON;
        
    ------------------------------------------------------------
    -- 0) Resolve job once
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#Job') IS NOT NULL DROP TABLE #Job;

    SELECT j.id AS JobId
    INTO #Job
    FROM dbo.t_jobs j
    WHERE j.jobnumber = @JobNumber;

    IF NOT EXISTS (SELECT 1 FROM #Job)
    BEGIN
        SELECT TOP 0 *
        FROM (VALUES
            (CAST(NULL AS NVARCHAR(4000)), CAST(NULL AS NVARCHAR(4000)), CAST(NULL AS NVARCHAR(4000)),
            CAST(NULL AS NVARCHAR(4000)), CAST(NULL AS NVARCHAR(4000)), CAST(NULL AS NVARCHAR(4000)),
            CAST(NULL AS NVARCHAR(4000)), CAST(NULL AS DECIMAL(18,2)), CAST(NULL AS DECIMAL(18,2)),
            CAST(NULL AS DECIMAL(18,2)), CAST(NULL AS DECIMAL(18,2)), CAST(NULL AS DECIMAL(18,2)),
            CAST(NULL AS DECIMAL(18,2)), CAST(NULL AS INT), CAST(NULL AS INT))
        ) v([WO #],[InvNo],[Subcontractor],[Category],[Description],[WO Type],[WO Status],
            [SOW Cost],[WO Cost],[WO Sell],[$ AP ex GST],[Residual Markup],[Cost Var],[RowOrder],[OrderGroup]);
        RETURN;
    END;

    ------------------------------------------------------------
    -- 1) EstimateItems -> #EstimateItems (cast itemid to NVARCHAR(36) so it can be indexed)
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#EstimateItems') IS NOT NULL DROP TABLE #EstimateItems;

    SELECT
        CAST(esi.itemid AS NVARCHAR(36)) AS itemid,
        esc.label AS [Category],
        esi.trade,
        CASE
            WHEN esi.materialCost <> 0 THEN esi.materialTotal
            WHEN esi.labourCost   <> 0 THEN esi.labourTotal
            ELSE 0
        END AS SOWCost
    INTO #EstimateItems
    FROM dbo.[t_estimates-snapshot-items]      esi
    JOIN dbo.[t_estimates-snapshot-categories] esc ON esc.categoryId = esi.categoryId
    JOIN dbo.[t_estimates-snapshot]            es  ON es.id = esc.estimateId
    JOIN #Job                                  jb  ON jb.JobId = es.jobid
    where es.estimateStatus in ('Variation Authorised','Authorised');    --add this to fix the authorised items issue in scopes

    CREATE INDEX IX_EstimateItems_itemid ON #EstimateItems(itemid);

    ------------------------------------------------------------
    -- 2) Pre-aggregate AP invoices once per (JobId, WorkOrderId)
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#APAgg') IS NOT NULL DROP TABLE #APAgg;

    SELECT
        ap.jobid,
        ap.workOrderId,
        STRING_AGG(ap.invoiceNumber, ' | ') AS InvNo,
        ROUND(SUM(ap.subtotal), 2)          AS APSubtotal
    INTO #APAgg
    FROM dbo.[t_accounts-payable-invoices] ap
    JOIN #Job jb ON jb.JobId = ap.jobid
    WHERE ap.accountsPayableInvoiceStatus NOT IN ('draft','cancelled','pending approval')
    GROUP BY ap.jobid, ap.workOrderId;

    CREATE INDEX IX_APAgg_job_wo ON #APAgg(jobid, workOrderId);

    ------------------------------------------------------------
    -- 3) WOType (woi.id = WorkOrderId per your schema)
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#WOType') IS NOT NULL DROP TABLE #WOType;

    SELECT
        CAST(woi.id AS NVARCHAR(36)) AS WorkOrderId,
        CASE
            WHEN COUNT(CASE WHEN woi.[ps/pc] IS NOT NULL THEN 1 END) = 0 THEN 'Fixed'
            WHEN COUNT(CASE WHEN woi.[ps/pc] IS NULL     THEN 1 END) > 0
            AND COUNT(CASE WHEN woi.[ps/pc] IS NOT NULL THEN 1 END) > 0 THEN 'Mixed'
            ELSE 'ps/pc'
        END AS WOType
    INTO #WOType
    FROM dbo.[t_work-orders-items] woi
    JOIN #Job jb ON jb.JobId = woi.jobId
    GROUP BY CAST(woi.id AS NVARCHAR(36));

    CREATE INDEX IX_WOType_wo ON #WOType(WorkOrderId);

    ------------------------------------------------------------
    -- 4) HeaderSOW (join wo.id = woi.id per your schema)
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#HeaderSOW') IS NOT NULL DROP TABLE #HeaderSOW;

    SELECT
        wo.id AS WorkOrderId,
        CAST(SUM(COALESCE(ei.SOWCost, 0)) AS DECIMAL(18,2)) AS TotalSOWCost
    INTO #HeaderSOW
    FROM dbo.[t_work-orders]       wo
    JOIN #Job                      jb  ON jb.JobId = wo.jobid
    JOIN dbo.[t_work-orders-items] woi ON woi.id = wo.id
    LEFT JOIN #EstimateItems       ei  ON ei.itemid = CAST(woi.estimateItemId AS NVARCHAR(36))
    WHERE wo.workOrderType = 'Authorised Works'
    GROUP BY wo.id;

    CREATE INDEX IX_HeaderSOW_wo ON #HeaderSOW(WorkOrderId);

    ------------------------------------------------------------
    -- 5) HeaderRows
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#HeaderRows') IS NOT NULL DROP TABLE #HeaderRows;

    SELECT
        CAST(wo.[label] AS NVARCHAR(4000))                 AS [WO #],
        CAST(ap.InvNo AS NVARCHAR(4000))                   AS [InvNo],
        CAST(c.name AS NVARCHAR(4000))                     AS [Subcontractor],
        CAST(NULL AS NVARCHAR(4000))                       AS [Category],
        CAST(NULL AS NVARCHAR(4000))                       AS [Description],
        CAST(wot.WOType AS NVARCHAR(4000))                 AS [WO Type],
        CAST(wo.workOrderStatus AS NVARCHAR(4000))         AS [WO Status],
        CAST(hs.TotalSOWCost AS DECIMAL(18,2))             AS [SOW Cost],
        CAST(wo.costTotal AS DECIMAL(18,2))                AS [WO Cost],
        CAST(wo.sellTotal AS DECIMAL(18,2))                AS [WO Sell],
        CAST(ap.APSubtotal AS DECIMAL(18,2))               AS [$ AP ex GST],
        CASE
            WHEN wo.workOrderStatus = 'Cancelled' THEN 0
                ELSE CAST(hs.TotalSOWCost - wo.costTotal AS DECIMAL(18,2))
        END AS [Residual Markup],
        --CAST(hs.TotalSOWCost - CAST(wo.costTotal AS DECIMAL(18,2)) AS DECIMAL(18,2)) AS [Residual Markup],
        CAST(ISNULL(ap.APSubtotal, 0) - wo.costTotal  AS DECIMAL(18,2))               AS [Cost Var],
        CAST(0 AS INT)                                     AS [RowOrder],
        CAST(0 AS INT)                                     AS [OrderGroup]
    INTO #HeaderRows
    FROM dbo.[t_work-orders] wo
    JOIN #Job jb            ON jb.JobId = wo.jobid
    JOIN dbo.t_contacts c   ON c.id = wo.assignedId
    LEFT JOIN #WOType wot   ON wot.WorkOrderId = wo.id
    LEFT JOIN #HeaderSOW hs ON hs.WorkOrderId  = wo.id
    LEFT JOIN #APAgg ap     ON ap.jobid = wo.jobid AND ap.workOrderId = wo.id
    WHERE wo.workOrderType = 'Authorised Works';

    ------------------------------------------------------------
    -- 6) LineRows (join wo.id = woi.id per your schema)
    ------------------------------------------------------------
    IF OBJECT_ID('tempdb..#LineRows') IS NOT NULL DROP TABLE #LineRows;

    SELECT
        CAST(wo.[label] AS NVARCHAR(4000))                                  AS [WO #],
        CAST(NULL AS NVARCHAR(4000))                                        AS [InvNo],
        CAST(NULL AS NVARCHAR(4000))                                        AS [Subcontractor],
        CAST(ei.[Category] AS NVARCHAR(4000))                               AS [Category],
        CAST(woi.[description] AS NVARCHAR(4000))                           AS [Description],
        CAST(woi.[ps/pc] AS NVARCHAR(4000))                                 AS [WO Type],
        CAST(NULL AS NVARCHAR(4000))                                        AS [WO Status],
        CAST(COALESCE(ei.SOWCost, 0) AS DECIMAL(18,2))                      AS [SOW Cost],
        CAST(woi.materialCostTotal + woi.labourCostTotal AS DECIMAL(18,2))  AS [WO Cost],
        CAST(woi.materialSellTotal + woi.labourSellTotal AS DECIMAL(18,2))  AS [WO Sell],
        CAST(NULL AS DECIMAL(18,2))                                         AS [$ AP ex GST], 
        CAST(NULL AS DECIMAL(18,2))                                         AS [Residual Markup],
        CAST(NULL AS DECIMAL(18,2))                                         AS [Cost Var],
        CAST(1 AS INT)                                                      AS [RowOrder],
        CAST(1 AS INT)                                                      AS [OrderGroup]
    INTO #LineRows
    FROM dbo.[t_work-orders-items] woi
    JOIN dbo.[t_work-orders] wo ON wo.id = woi.id
    JOIN #Job jb               ON jb.JobId = woi.jobid
    LEFT JOIN #EstimateItems ei ON ei.itemid = CAST(woi.estimateItemId AS NVARCHAR(36))
    WHERE wo.workOrderType = 'Authorised Works';

    ------------------------------------------------------------
    -- 7) Final output
    ------------------------------------------------------------
    SELECT *
    FROM (
        SELECT * FROM #HeaderRows
        UNION ALL
        SELECT * FROM #LineRows
    ) allrows
    ORDER BY
        [WO #],
        RowOrder,
        CASE
            WHEN LOWER(Category) LIKE '%general%' THEN 1
            WHEN Category = 'Preliminaries'       THEN 2
            WHEN Category = 'Insurances'           THEN 3
            WHEN LOWER(Category) LIKE '%xx%'       THEN 4
            WHEN LOWER(Category) LIKE '%qbcc%'     THEN 5
            ELSE 0
        END,
        Category,
        [Description];
END;
GO


**14** table function to return required data from the job for the finance finals workbook

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE FUNCTION [reporting].[FinalsJobData] (@JobNumber NVARCHAR(50))
RETURNS TABLE
AS
RETURN
(
    WITH AuthTotals AS (
        SELECT 
            j.jobNumber,
            SUM(esi.[labourTotal] + esi.materialTotal) AS AuthWorks
        FROM [dbo].[t_estimates-snapshot-items] esi
        JOIN [dbo].[t_estimates-snapshot-categories] esc 
            ON esi.categoryId = esc.categoryId
        JOIN [t_estimates-snapshot] es 
            ON es.id = esc.estimateId
        JOIN t_jobs j 
            ON j.id = es.jobid
        WHERE es.estimateType = 'Authorised Works'
          AND es.estimateStatus LIKE '%Auth%'
          AND esi.trade NOT IN ('Supervision','Supervisor', 'Home Warranty Insurance', 'QBCC')
          AND NOT (
                (esi.description LIKE '%q%leave%' OR esi.description LIKE '%qbcc%' OR esi.description LIKE '%supervision%') 
                AND esc.label IN ('general','insurances','premilinaries','xx','QBCC','Supervision')
          )
          AND NOT (
                (esi.description LIKE '%event%levy%' OR esi.description LIKE '%event%loading%' 
                 OR esi.description LIKE '%cat%levy%' OR esi.description LIKE '%cat%loading%') 
                AND esc.label IN ('general','insurances','premilinaries','xx')
          )
          AND j.jobnumber = @JobNumber
        GROUP BY j.jobNumber
    ),
    ExcessAR AS (
        SELECT
            j.jobnumber,
            SUM(CASE WHEN xari.LineAmount > 0 THEN xari.LineAmount ELSE 0 END) AS ExcessPaid_AR,
            SUM(CASE WHEN xari.LineAmount < 0 THEN -xari.LineAmount ELSE 0 END) AS ExcessRefund_AR
        FROM [xero].[AR-invoices-Items] xari
        JOIN [xero].[AR-invoices] xar ON xar.InvoiceID = xari.InvoiceID
        LEFT JOIN dbo.[t_accounts-receivable-invoices] ar ON ar.[InvoiceNumber] = xar.InvoiceNumber
        LEFT JOIN dbo.[t_accounts-payable-invoices]   ap ON ap.invoiceNumber = xar.InvoiceNumber
        LEFT JOIN t_jobs j ON j.id = COALESCE(ar.jobid, ap.jobid)
        WHERE xari.AccountCode IN ('230','834')
          AND xar.[Status] = 'paid'
          AND j.jobnumber = @JobNumber
        GROUP BY j.jobnumber
    ),
    ExcessAP AS (
        SELECT
            j.jobnumber,
            SUM(xapi.LineAmount) AS ExcessRefund_AP
        FROM [xero].[Ap-invoices-Items] xapi
        JOIN [xero].[Ap-invoices] xap ON xap.InvoiceID = xapi.InvoiceID
        LEFT JOIN dbo.[t_accounts-payable-invoices] ap ON ap.invoiceNumber = xap.InvoiceNumber
        LEFT JOIN dbo.[t_accounts-receivable-invoices] ar ON ar.invoiceNumber = xap.InvoiceNumber
        LEFT JOIN t_jobs j ON j.id = COALESCE(ap.jobid, ar.jobid)
        WHERE xapi.AccountCode IN ('230','834')
          AND xap.[Status] = 'paid'
          AND j.jobnumber = @JobNumber
        GROUP BY j.jobnumber
    )
    SELECT 
       CONVERT(VARCHAR(50),  ISNULL(j.clientReference, ''))                 AS clientReference,
        CONVERT(VARCHAR(200), ISNULL(j.addressLine1, ''))                    AS [Address],
        CONVERT(VARCHAR(100), ISNULL(c.name, ''))                            AS [Instructing Client],
        CONVERT(VARCHAR(100), ISNULL(cl.[Client Group], ''))                 AS [Client Group],
        CONVERT(VARCHAR(20),  CASE
                                 WHEN cl.[Client Group] IN ('Hollard','IAG','RACQ') THEN 'Cost +'
                                 ELSE 'As Quoted'
                               END)                                          AS [Basis],
        CONVERT(VARCHAR(100), ISNULL(u.fullname, ''))                        AS [Case Manager],
        CONVERT(VARCHAR(20),  CASE 
                                 WHEN LOWER(ctc.code) LIKE 'bau%' 
                                   OR LOWER(ctc.code) LIKE 'business as usual%' 
                                   OR LOWER(ctc.code) LIKE 'no-cat%' 
                                   THEN 'BAU' 
                                 ELSE REPLACE(REPLACE(LEFT(ctc.code, 7), ' ', ''), '-', '') 
                               END)                                          AS CatCode,
        /* numeric fields left as-is for ADO */
        (CASE 
            WHEN cl.[Client Group] IN ('Suncorp', 'AAI')
                AND j.CreatedAt > '2025-10-01'
            THEN 0.16
            ELSE
                (SELECT TOP 1 clm.Markup
                    FROM [Clients-Markup] clm
                    WHERE clm.[Client Group] = cl.[Client Group]
                    AND a.AuthWorks > clm.MinTier
                    AND a.AuthWorks <= clm.MaxTier
                    ORDER BY clm.MinTier DESC
                ) + COALESCE(cc.[MU Loading], 0)
         END) AS Markup, 
        COALESCE(cc.[catlevy],0)                                             AS CatLevy,
        (SELECT SUM(ar.subtotal)
           FROM [t_accounts-receivable-invoices] ar
          WHERE ar.accountsReceivableInvoiceStatus NOT IN ('draft', 'cancelled')
            AND ar.accountNumber NOT IN ('212','213','214','230','834') 
            AND ar.jobid = j.id
          GROUP BY ar.jobid)                                                 AS [Progress],
        COALESCE(ear.ExcessPaid_AR, 0)                                       AS [ExcessPaid],
        (COALESCE(ear.ExcessRefund_AR, 0) + COALESCE(eap.ExcessRefund_AP, 0)) AS [ExcessRefund],
        (select cast(max(sh.createdAt) as date) 
        from [t_status-histories] sh 
        join t_statuses s on s.id = sh.newstatusid 
        where sh.jobid = j.id and s.statusType ='Accounts' and s.name = 'Ready To Invoice') as [Completion],
        cast(j.createdAt as date) as [CreatedAt]
    FROM [t_jobs] j 
    LEFT JOIN t_contacts c ON j.clientId = c.id 
    LEFT JOIN t_users u ON u.id = j.caseManagerId
    LEFT JOIN clients cl ON cl.[Instructing Client] = c.name 
    LEFT JOIN [t_catastrophe-codes] ctc ON ctc.id = j.catastropheCodeId 
    LEFT JOIN [Clients-CatLevy] cc 
        ON cc.[Client Group] = cl.[Client Group] 
       AND cc.CatCode = CASE 
                            WHEN LOWER(ctc.code) LIKE 'bau%'  
                              OR LOWER(ctc.code) LIKE 'business as usual%'  
                              OR LOWER(ctc.code) LIKE 'no-cat%' THEN 'BAU' 
                            ELSE REPLACE(REPLACE(LEFT(ctc.code, 7), ' ', ''), '-', '') 
                        END 
    JOIN AuthTotals a ON a.jobNumber = j.jobNumber
    LEFT JOIN ExcessAR ear ON ear.jobnumber = j.jobnumber
    LEFT JOIN ExcessAP eap ON eap.jobnumber = j.jobnumber
    WHERE j.jobnumber = @JobNumber
);
GO


**15** table function to return estimate data for a specific job for the finance final invoicing workbook

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE FUNCTION [reporting].[FinalsSOWData] (@JobNumber NVARCHAR(50))
RETURNS TABLE
AS
RETURN

SELECT
    CAST(ISNULL(es.label, '') AS NVARCHAR(4000))                    AS [Estimate],
    CAST(woi.[label] AS NVARCHAR(4000))                             AS [WorkOrder],
    CASE
        WHEN esc.label IS NOT NULL AND LOWER(esc.label) LIKE '%provisional sum%' THEN 'PS Allow'
        WHEN esc.label IS NOT NULL AND LOWER(esc.label) LIKE '%prime cost%'       THEN 'PC Allow'
        WHEN esi.[trade] IS NOT NULL AND LOWER(esi.[trade]) LIKE '%provisional sum%' THEN 'PS Allow'
        WHEN esi.[trade] IS NOT NULL AND LOWER(esi.[trade]) LIKE '%prime cost%'      THEN 'PC Allow'
        WHEN esi.[description] IS NOT NULL AND (
             LOWER(esi.[description]) LIKE '%pc allow%' OR
             LOWER(esi.[description]) LIKE '%prime cost%' OR
             LOWER(esi.[description]) LIKE '%pc cost%' OR
             LOWER(esi.[description]) LIKE '%ps allow%' OR
             LOWER(esi.[description]) LIKE '%provisional sum%'
        ) THEN
            CASE WHEN LOWER(esi.[description]) LIKE '%pc%' THEN 'PC Allow' ELSE 'PS Allow' END
        WHEN (esi.[materialUnit] = 'PC' OR esi.[labourUnit] = 'PC') THEN 'PC Allow'
        WHEN (esi.[materialUnit] = 'PS' OR esi.[labourUnit] = 'PS') THEN 'PS Allow'
        ELSE NULL
    END                                                             AS [ps/pc],

    CAST(ISNULL(esc.label, '') AS NVARCHAR(4000))                   AS [Category],
    CAST(ISNULL(esc.[description], '') AS NVARCHAR(4000))           AS [Dimensions],
    esi.[authorised],
    CAST(ISNULL(esi.[trade], '') AS NVARCHAR(4000))                 AS [Trade Summary],
    CAST(ISNULL(esi.[description], '') AS NVARCHAR(4000))           AS [Description],

    CASE WHEN esi.[materialCost] <> 0 THEN esi.[materialQuantity]
         WHEN esi.[labourCost]   <> 0 THEN esi.[labourQuantity]
    END                                                             AS [Qty],

    CAST(CASE WHEN esi.[materialCost] <> 0 THEN ISNULL(esi.[materialUnit], '')
              WHEN esi.[labourCost]   <> 0 THEN ISNULL(esi.[labourUnit],  '')
         END AS NVARCHAR(4000))                                     AS [Unit],

    CASE WHEN esi.[materialCost] <> 0 THEN esi.[materialCost]
         WHEN esi.[labourCost]   <> 0 THEN esi.[labourCost]
    END                                                             AS [Rate],

    CASE WHEN esi.[materialCost] <> 0 THEN esi.[materialTotal]
         WHEN esi.[labourCost]   <> 0 THEN esi.[labourTotal]
    END                                                             AS [Amount],

    esi.tax                                                         AS [Tax],

    CASE WHEN esi.[materialCost] <> 0 THEN ROUND(esi.materialMargin, 2)
         WHEN esi.[labourCost]   <> 0 THEN ROUND(esi.labourMargin,  2)
    END                                                             AS [Markup%],

    CASE WHEN esi.[materialCost] <> 0 THEN ROUND(esi.[materialTotal] * esi.materialMargin, 2)
         WHEN esi.[labourCost]   <> 0 THEN ROUND(esi.[labourTotal]  * esi.labourMargin,  2)
    END                                                             AS [Markup]
FROM [dbo].[t_estimates-snapshot-items]          AS esi
JOIN [dbo].[t_estimates-snapshot-categories]     AS esc ON esc.categoryId = esi.categoryId
JOIN [dbo].[t_estimates-snapshot]                AS es  ON es.id = esi.estimateId
JOIN [dbo].[t_jobs]                              AS j   ON j.id = es.jobId
LEFT JOIN [dbo].[t_work-orders-items]            AS woi ON woi.estimateItemId = esi.itemid
WHERE j.jobnumber = @JobNumber
  AND es.estimateType = 'Authorised Works'
  AND es.estimateStatus in ('Authorised','Variation Authorised');
GO

--jobs status
WITH status_history_ranked AS (
    SELECT 
        sh.jobid,
        sh.newstatusid,
        sh.createdAt,
        s.name AS status,
        s.statusType,
        ROW_NUMBER() OVER (
            PARTITION BY sh.jobid 
            ORDER BY sh.createdAt DESC
        ) AS rn
    FROM [t_status-histories] sh
    JOIN t_statuses s ON s.id = sh.newstatusid
    WHERE sh.createdAt < '2025-07-01'
),

latest_status_per_job AS (
    SELECT 
        jobid,
        status,
        statusType,
        createdAt AS statusDate
    FROM status_history_ranked
    WHERE rn = 1
)

SELECT 
    j.jobnumber,
    j.clientReference,
    ls.statusDate AS [Date Invoiced],
    ls.status,
    ls.statusType
from t_jobs j
LEFT JOIN latest_status_per_job ls ON ls.jobid = j.id
